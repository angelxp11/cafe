        && getAfter(/databases/$(database)/documents/facturas/$(request.resource.data.ultimaFacturaId)).data.turnoId == getAfter(/databases/$(database)/documents/configuracion/turnoActual).data.turnoId)
        || (get(/databases/$(database)/documents/usuarios/$(request.auth.token.email)).data.rol == 'admin'
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['saldoPendiente', 'estado', 'actualizadaEn'])
          && request.resource.data.saldoPendiente is number
          && request.resource.data.saldoPendiente >= 0
          && request.resource.data.saldoPendiente <= resource.data.total
          && request.resource.data.actualizadaEn is timestamp
          && ((request.resource.data.estado == 'cancelada'
              && request.resource.data.saldoPendiente == 0
              && getAfter(/databases/$(database)/documents/facturas/$(cuentaId)).data.cancelada == true)
            || (request.resource.data.estado == 'pendiente'
              && request.resource.data.saldoPendiente == resource.data.saldoPendiente + getAfter(/databases/$(database)/documents/facturas/$(resource.data.ultimaFacturaId)).data.monto
              && getAfter(/databases/$(database)/documents/facturas/$(resource.data.ultimaFacturaId)).data.cancelada == true
              && getAfter(/databases/$(database)/documents/facturas/$(resource.data.ultimaFacturaId)).data.cuentaClienteId == cuentaId))));
      allow delete: if false;
    }

    match /categorias/{categoriaId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
        && get(/databases/$(database)/documents/usuarios/$(request.auth.token.email)).data.rol == 'admin';
    }

    match /date-expired/{documentId} {
      allow get: if documentId == 'date-expired'
        && resource.data['date-expired'] is timestamp;
      allow list, create, update, delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
